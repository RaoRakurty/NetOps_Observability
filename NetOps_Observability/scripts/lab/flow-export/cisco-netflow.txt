! Cisco Cat8000v IOS-XE 17.06 (lan-sw1 172.40.40.51, lan-sw2 172.40.40.52)
! Flexible NetFlow v9 export to the NetOps collector.
! UNVERIFIED: vrnetlab VM; could not audit running config from the host.
! SSH to the mgmt IP with device creds, "configure terminal", paste.
! mgmt is GigabitEthernet1; replace GigabitEthernet2 with the data interface(s).

flow record NETOPS-REC
 match ipv4 source address
 match ipv4 destination address
 match ipv4 protocol
 match transport source-port
 match transport destination-port
 match ipv4 tos
 collect counter bytes long
 collect counter packets long
 collect interface input
 collect interface output
!
flow exporter NETOPS-EXP
 destination 172.40.40.122
 source GigabitEthernet1
 transport udp 2055
 export-protocol netflow-v9
 template data timeout 30
 option interface-table timeout 60
!
flow monitor NETOPS-MON
 exporter NETOPS-EXP
 record NETOPS-REC
 cache timeout active 60
 cache timeout inactive 15
!
interface GigabitEthernet2
 ip flow monitor NETOPS-MON input
 ip flow monitor NETOPS-MON output
!
! end / write memory
! Verify:  show flow exporter NETOPS-EXP statistics ; show flow monitor NETOPS-MON cache
!
